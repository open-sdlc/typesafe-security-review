public List<Order> findOrdersForCustomer(String customerName) throws SQLException {
    String sql = "SELECT id, total, status FROM orders WHERE customer_name = ?";
    try (PreparedStatement stmt = connection.prepareStatement(sql)) {
        stmt.setString(1, customerName);
        try (ResultSet rs = stmt.executeQuery()) {
            List<Order> orders = new ArrayList<>();
            while (rs.next()) {
                orders.add(new Order(
                    rs.getLong("id"), rs.getBigDecimal("total"), rs.getString("status")));
            }
            return orders;
        }
    }
}
