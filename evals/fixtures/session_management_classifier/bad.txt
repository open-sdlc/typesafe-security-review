import time

_session_counter = 0

def create_session_id(user_id):
    global _session_counter
    _session_counter += 1
    timestamp = int(time.time())
    # combine timestamp and an incrementing counter for uniqueness
    session_id = f"{timestamp}-{user_id}-{_session_counter}"
    return session_id
