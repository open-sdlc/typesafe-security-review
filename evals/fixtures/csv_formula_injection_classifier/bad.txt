import csv

def export_contacts(contacts, response):
    writer = csv.writer(response)
    writer.writerow(["Name", "Email", "Notes"])
    for contact in contacts:
        # contact.notes is a free-text field a user filled in on their profile
        writer.writerow([contact.name, contact.email, contact.notes])
    return response
