def download_firmware_update(update_url):
    response = requests.get(update_url, verify=True)  # https://updates.example-motors.com/firmware/latest.bin
    firmware_path = "/tmp/firmware_latest.bin"
    with open(firmware_path, "wb") as f:
        f.write(response.content)
    signature = requests.get(update_url + ".sig", verify=True).content
    if not verify_signature(firmware_path, signature, VENDOR_PUBLIC_KEY):
        raise FirmwareIntegrityError("Signature verification failed; refusing to flash")
    return firmware_path

def apply_firmware_update(firmware_path):
    flash_ecu(firmware_path)
    log.info("Firmware update applied")
