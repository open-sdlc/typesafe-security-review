function renderComment(comment) {
  // Strip <script> tags before rendering user comments
  const cleaned = comment.replace(/<script.*?>.*?<\/script>/gi, '');
  document.getElementById('comments').innerHTML += `<div class="comment">${cleaned}</div>`;
}
