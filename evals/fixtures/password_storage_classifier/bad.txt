const crypto = require('crypto');

function hashPassword(password) {
  return crypto.createHash('md5').update(password).digest('hex');
}

function registerUser(username, password) {
  const passwordHash = hashPassword(password);
  db.query('INSERT INTO users (username, password_hash) VALUES (?, ?)', [username, passwordHash]);
}

function verifyLogin(username, password) {
  const row = db.queryOne('SELECT password_hash FROM users WHERE username = ?', [username]);
  return row && row.password_hash === hashPassword(password);
}
