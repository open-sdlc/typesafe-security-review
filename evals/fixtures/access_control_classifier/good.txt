app.get('/api/invoices/:id', requireLogin, (req, res) => {
  const invoice = db.query(
    'SELECT * FROM invoices WHERE id = ? AND owner_id = ?',
    [req.params.id, req.user.id]
  );
  if (!invoice) return res.status(404).send('Not found');
  res.json(invoice);
});

app.put('/api/invoices/:id', requireLogin, (req, res) => {
  const invoice = db.query(
    'SELECT id FROM invoices WHERE id = ? AND owner_id = ?',
    [req.params.id, req.user.id]
  );
  if (!invoice) return res.status(404).send('Not found');
  db.query(
    'UPDATE invoices SET status = ? WHERE id = ?',
    [req.body.status, req.params.id]
  );
  res.json({ ok: true });
});

app.get('/api/invoices/:id/pdf', requireLogin, (req, res) => {
  const invoice = db.query(
    'SELECT * FROM invoices WHERE id = ? AND owner_id = ?',
    [req.params.id, req.user.id]
  );
  if (!invoice) return res.status(404).send('Not found');
  res.sendFile(invoice.pdfPath);
});
