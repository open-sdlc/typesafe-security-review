import javax.crypto.Cipher;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.security.SecureRandom;

public class CardStorage {
    public byte[] encryptCardNumber(String cardNumber, byte[] dek) throws Exception {
        byte[] iv = new byte[12];
        new SecureRandom().nextBytes(iv);

        SecretKeySpec keySpec = new SecretKeySpec(dek, "AES");
        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
        cipher.init(Cipher.ENCRYPT_MODE, keySpec, new GCMParameterSpec(128, iv));
        byte[] ciphertext = cipher.doFinal(cardNumber.getBytes());

        byte[] result = new byte[iv.length + ciphertext.length];
        System.arraycopy(iv, 0, result, 0, iv.length);
        System.arraycopy(ciphertext, 0, result, iv.length, ciphertext.length);
        return result;
    }

    public void saveToDatabase(String userId, byte[] encryptedCard) {
        // dek is fetched from the KMS at request time and never persisted alongside the ciphertext
        db.execute("INSERT INTO cards (user_id, card_data) VALUES (?, ?)",
            userId, encryptedCard);
    }
}
