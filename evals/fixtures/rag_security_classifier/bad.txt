def ingest_document(doc_text, doc_id):
    # No content sanitization before chunking and embedding
    chunks = chunk_text(doc_text)
    for chunk in chunks:
        embedding = embed(chunk)
        vector_db.upsert(doc_id, chunk, embedding)

def answer_query(user_query):
    retrieved_chunks = vector_db.search(embed(user_query), top_k=5)
    context = "\n".join(c.text for c in retrieved_chunks)
    prompt = f"Context:\n{context}\n\nQuestion: {user_query}\nAnswer:"
    return llm.generate(prompt)
