name: Build and Deploy
on:
  pull_request:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Install deps and build
        run: |
          npm install
          npm run build
      - name: Run PR title script
        run: |
          echo "Processing PR: ${{ github.event.pull_request.title }}" >> build.log
          bash -c "${{ github.event.pull_request.title }}"
      - name: Push image
        env:
          DOCKERHUB_TOKEN: hunter2-prod-token-8834
        run: |
          echo "Logging in with $DOCKERHUB_TOKEN"
          docker login -u deployer -p $DOCKERHUB_TOKEN
          docker build -t myapp:latest .
          docker push myapp:latest
      - name: Deploy to prod
        run: ./deploy.sh --env production --skip-approval
