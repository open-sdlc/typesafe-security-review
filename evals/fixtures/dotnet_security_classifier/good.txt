public class InvoiceController : Controller
{
    private readonly string _connectionString = ConfigurationManager.ConnectionStrings["Default"].ConnectionString;

    [HttpGet]
    [Authorize]
    public ActionResult Search(string customerName)
    {
        var results = new List<Invoice>();
        using (var conn = new SqlConnection(_connectionString))
        {
            conn.Open();
            const string query = "SELECT * FROM Invoices WHERE CustomerName = @CustomerName";
            var cmd = new SqlCommand(query, conn);
            cmd.Parameters.Add("@CustomerName", SqlDbType.NVarChar, 200).Value = customerName;
            using (var reader = cmd.ExecuteReader())
            {
                while (reader.Read())
                {
                    results.Add(new Invoice
                    {
                        Id = (int)reader["Id"],
                        Amount = (decimal)reader["Amount"]
                    });
                }
            }
        }
        return View(results);
    }
}
