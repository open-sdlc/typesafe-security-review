<script>
  function loadWidget() {
    var apiUrl = window.defaultConfig.apiUrl;
    var script = document.createElement("script");
    script.src = apiUrl + "/widget.js";
    document.body.appendChild(script);
  }
  loadWidget();
</script>

</div>
